-- --- SMOOTH BOUNCY MENU ANIMATION LOGIC ---
local isOpen = false
local isAnimating = false

local function AnimateMenu(open)
    if isAnimating then return end
    isAnimating = true

    if open then
        MainWrapper.Visible = true
        local scaleTween = TweenService:Create(MainScale, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1})
        scaleTween:Play()
        scaleTween.Completed:Wait()
        isAnimating = false
    else
        local scaleTween = TweenService:Create(MainScale, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.In), {Scale = 0})
        scaleTween:Play()
        scaleTween.Completed:Wait()
        MainWrapper.Visible = false
        isAnimating = false
    end
end

OpenBtn.MouseButton1Click:Connect(function()
    isOpen = not isOpen
    AnimateMenu(isOpen)
end)

CloseBtn.MouseButton1Click:Connect(function()
    isOpen = false
    AnimateMenu(false)
end)

local GodmodeActive = true
local ConnectionPool = {}

GodmodeButton.MouseButton1Click:Connect(function()
    GodmodeActive = not GodmodeActive
    if GodmodeActive then
        GodmodeButton.Text = "BẤT TỬ: ON"
        GodmodeButton.TextColor3 = Color3.fromRGB(0, 255, 128)
    else
        GodmodeButton.Text = "BẤT TỬ: OFF"
        GodmodeButton.TextColor3 = Color3.fromRGB(255, 50, 50)
    end
end)

local function secureCharacter(character)
    if not character then return end
    local humanoid = character:WaitForChild("Humanoid", 5)
    local rootPart = character:WaitForChild("HumanoidRootPart", 5)
    if not humanoid or not rootPart then return end

    local heartbeatConn = RunService.Heartbeat:Connect(function()
        if GodmodeActive then
            pcall(function()
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
                humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
                humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
                humanoid:SetStateEnabled(Enum.HumanoidStateType.PlatformStanding, false)
                humanoid.Health = humanoid.MaxHealth

                for _, obj in ipairs(Workspace:GetDescendants()) do
                    if obj:IsA("BasePart") and not obj:IsDescendantOf(character) then
                        local nameLower = obj.Name:lower()
                        if nameLower:find("kill") or nameLower:find("laser") or nameLower:find("death") or nameLower:find("damage") or nameLower:find("hazard") then
                            obj.CanTouch = false
                            obj.CanCollide = false
                        end
                    end
                end
            end)
        end
    end)
    table.insert(ConnectionPool, heartbeatConn)
end

if LocalPlayer.Character then
    task.spawn(function() secureCharacter(LocalPlayer.Character) end)
end

LocalPlayer.CharacterAdded:Connect(function(newChar)
    for _, conn in ipairs(ConnectionPool) do
        pcall(function() conn:Disconnect() end)
    end
    ConnectionPool = {}
    task.spawn(function() secureCharacter(newChar) end)
end)
